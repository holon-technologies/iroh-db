232    pub fn operational_metrics(&self) -> Result<OperationalMetrics, DbError> {
233        let runtime = self.inner.telemetry.snapshot();
234        let (active_migrations, staged_migration_records, frozen_migrations) =
235            self.inner.store.migration_counts(self.domain_id())?;
236        let snapshot_age = self
237            .inner
238            .store
239            .snapshot_observed_at(self.domain_id())?

242                    .duration_since(UNIX_EPOCH)
243                    .unwrap_or_default()
244                    .as_secs()
245                    .saturating_sub(observed)
246            })
247            .map(Duration::from_secs);
248        Ok(OperationalMetrics {

255            materialized_records: self

282    /// Verifies content hashes, signatures, encryption, dependency presence, and rebuild equality.
283    #[tracing::instrument(name = "iroh_db.verify", skip_all)]
284    pub async fn verify(&self) -> Result<VerificationReport, DbError> {
285        let _control_reader = self.inner.control_gate.read().await;
286        let _writer = self.inner.writer.lock().await;
287        self.verify_unlocked().await
288    }
289
290    async fn verify_unlocked(&self) -> Result<VerificationReport, DbError> {
291        if !self
292            .inner
293            .store
294            .authority_checkpoint_consistent(self.domain_id())?
295        {
296            return Err(DbError::Operation(
297                "authority checkpoint evidence is incomplete".into(),
298            ));
299        }
300        if !self
301            .inner
302            .store
303            .subject_revocations_consistent(self.domain_id())?
304        {
305            return Err(DbError::Operation(
306                "subject revocation index differs from signed control".into(),
307            ));
308        }
309        let commit_ids = self.inner.store.list_stored_commit_ids(self.domain_id())?;
310        for commit_id in &commit_ids {
311            let envelope = self.inner.store.load_commit(*commit_id).await?;
312            if envelope.header().domain_id() != self.domain_id() {
313                return Err(DbError::Operation("commit domain mismatch".into()));
314            }
315            let key = self
316                .inner
317                .credentials
318                .load_domain_epoch(self.domain_id(), envelope.header().epoch())?
319                .domain_key();
320            open_commit(&envelope, &key)?;
321            for dependency in envelope.header().dependencies() {
322                if !self
323                    .inner
324                    .store
325                    .contains_commit(self.domain_id(), *dependency)?
326                {
327                    return Err(DbError::Operation(format!(
328                        "commit dependency is absent: {dependency}"
329                    )));
330                }
331            }
332        }
333        let orphans = self.inner.store.scan_commit_orphans().await?;
334        let current = canonical_records(self.inner.store.export_materialized(self.domain_id())?)?;
335        let adapters_available = self.inner.has_record_adapters();
336        let no_replay_input = commit_ids.is_empty()
337            && self
338                .inner
339                .store
340                .snapshot_baseline(self.domain_id())?
341                .is_none();
342        let typed_rebuild_performed = adapters_available || no_replay_input;
343        let derived_consistent = if typed_rebuild_performed {
344            canonical_records(self.rebuild_mutations().await?)? == current
345        } else {
346            false
347        };
348        Ok(VerificationReport {
349            commits: commit_ids.len(),
350            orphan_commits: orphans.len(),
351            typed_rebuild_performed,
352            derived_consistent,
353            state_root: self.state_root()?,
354        })
355    }

357    /// Rebuilds typed materialized state in memory and swaps it in one metadata transaction.
358    #[tracing::instrument(name = "iroh_db.repair", skip_all)]
359    pub async fn repair(&self) -> Result<VerificationReport, DbError> {
360        let result = self.repair_inner().await;
361        self.inner.telemetry.record_repair(
362            result
363                .as_ref()
364                .ok()
365                .map(VerificationReport::derived_consistent),
366        );
367        result
368    }
369
370    async fn repair_inner(&self) -> Result<VerificationReport, DbError> {
371        let _control_reader = self.inner.control_gate.read().await;
372        let _writer = self.inner.writer.lock().await;
373        let rebuilt = self.rebuild_mutations().await?;
374        self.inner
375            .store
376            .repair_subject_revocations(self.domain_id())?;
377        self.inner
378            .store
379            .replace_materialized(self.domain_id(), &rebuilt)?;
380        self.verify_unlocked().await
381    }

410/// Creates an authenticated backup from a closed database using its default file key provider.
411#[tracing::instrument(name = "iroh_db.backup", skip_all)]
412pub fn backup_closed(
413    source: impl AsRef<Path>,
414    destination: impl AsRef<Path>,
415    options: BackupOptions,
416) -> Result<BackupReport, DbError> {
417    let source = source.as_ref();
418    let signer = LocalCredentials::load(source.join("keys"))?.signer();
419    backup_closed_with_signer(source, destination, options, &signer)
420}

422/// Creates a closed-store backup authenticated by an application-managed key provider.
423#[tracing::instrument(name = "iroh_db.backup", skip_all)]
424pub fn backup_closed_with_signer(
425    source: impl AsRef<Path>,
426    destination: impl AsRef<Path>,
427    options: BackupOptions,
428    signer: &iroh_db_security::IrohSigner,
429) -> Result<BackupReport, DbError> {
430    let source = source.as_ref();
431    let destination = destination.as_ref();
432    if !source.join("FORMAT").is_file() || destination.exists() {
433        return Err(operation_error(
434            "source is not a database or destination exists",
435        ));
436    }
437    let lock = OpenOptions::new()
438        .read(true)
439        .write(true)
440        .open(source.join("locks/writer.lock"))
441        .map_err(io_operation)?;
442    lock.try_lock_exclusive()
443        .map_err(|_| operation_error("database must be closed before backup"))?;
444    let mut files = Vec::new();
445    collect_files(source, source, options, &mut files)?;
446    files.sort();
447    if files.len() > MAX_CATALOG_ENTRIES {
448        return Err(operation_error("backup contains too many files"));
449    }
450    let staging = create_staging_directory(destination, "backup")?;
451    let result = build_backup(source, &staging, &files, signer);
452    let report = match result {
453        Ok(report) => {
454            publish_staging_directory(&staging, destination)?;
455            report
456        }
457        Err(error) => {
458            discard_staging_directory(&staging);
459            return Err(error);
460        }
461    };
462    drop(lock);
463    Ok(report)
464}

466fn build_backup(
467    source: &Path,
468    staging: &Path,
469    files: &[PathBuf],
470    signer: &iroh_db_security::IrohSigner,
471) -> Result<BackupReport, DbError> {
472    let mut entries = Vec::with_capacity(files.len());
473    let mut total = 0_u64;
474    for relative in files {
475        let target = staging.join(relative);
476        if let Some(parent) = target.parent() {
477            std::fs::create_dir_all(parent).map_err(io_operation)?;
478        }
479        let (hash, size) = copy_and_hash(&source.join(relative), &target)?;
480        total = total
481            .checked_add(size)
482            .ok_or_else(|| operation_error("backup byte count overflow"))?;
483        entries.push(BackupCatalogEntry {
484            hash,
485            size,
486            path: canonical_relative(relative)?,
487        });
488    }
489    entries.sort_by(|left, right| left.path.cmp(&right.path));
490    let content = BackupCatalogContent {
491        version: BACKUP_FORMAT_VERSION,
492        entries,
493    };
494    let content = minicbor::to_vec(content).map_err(backup_codec)?;
495    if u64::try_from(content.len()).unwrap_or(u64::MAX) > MAX_CATALOG_BYTES {
496        return Err(operation_error("backup catalog is too large"));
497    }
498    let catalog = SignedBackupCatalog {
499        signature: signer.sign_backup_catalog(&content),
500        signer: signer.author(),
501        content,
502    };
503    let catalog = minicbor::to_vec(catalog).map_err(backup_codec)?;
504    write_new_file(&staging.join(CATALOG_FILE), &catalog)?;
505    Ok(BackupReport {
506        files: files.len(),
507        bytes: total,
508        signer: signer.author(),
509    })
510}

524/// Restores only when the authenticated catalog signer matches the expected device.
525#[tracing::instrument(name = "iroh_db.restore", skip_all)]
526pub fn restore_backup_from(
527    backup: impl AsRef<Path>,
528    destination: impl AsRef<Path>,
529    expected_signer: AuthorId,
530) -> Result<BackupReport, DbError> {
531    restore_backup_impl(backup.as_ref(), destination.as_ref(), Some(expected_signer))
532}
533
534fn restore_backup_impl(
535    backup: &Path,
536    destination: &Path,
537    expected_signer: Option<AuthorId>,
538) -> Result<BackupReport, DbError> {
539    if destination.exists() {
540        return Err(operation_error("restore destination already exists"));
541    }
542    require_directory_without_symlink(backup)?;
543    let catalog_bytes =
544        read_bounded_regular_file(backup, Path::new(CATALOG_FILE), MAX_CATALOG_BYTES)?;
545    let catalog: SignedBackupCatalog = decode_backup_canonical(&catalog_bytes)?;
546    verify_backup_signature(catalog.signer, &catalog.content, &catalog.signature)?;
547    if expected_signer.is_some_and(|expected| expected != catalog.signer) {
548        return Err(operation_error(
549            "backup signer does not match the expected device",
550        ));
551    }
552    let content: BackupCatalogContent = decode_backup_canonical(&catalog.content)?;
553    validate_catalog(&content)?;
554    let mut total = 0_u64;
555    for entry in &content.entries {
556        let relative = Path::new(&entry.path);
557        let (hash, size) = hash_backup_file(backup, relative)?;
558        if size != entry.size || hash != entry.hash {
559            return Err(operation_error("backup checksum mismatch"));
560        }
561        total = total
562            .checked_add(size)
563            .ok_or_else(|| operation_error("restore byte count overflow"))?;
564    }
565    let staging = create_staging_directory(destination, "restore")?;
566    let result = restore_into_staging(backup, &staging, &content.entries);
567    if let Err(error) = result {
568        discard_staging_directory(&staging);
569        return Err(error);
570    }
571    publish_staging_directory(&staging, destination)?;
572    Ok(BackupReport {
573        files: content.entries.len(),
574        bytes: total,
575        signer: catalog.signer,
576    })
577}

579fn restore_into_staging(
580    backup: &Path,
581    staging: &Path,
582    entries: &[BackupCatalogEntry],
583) -> Result<(), DbError> {
584    for entry in entries {
585        let relative = Path::new(&entry.path);
586        let target = staging.join(relative);
587        if let Some(parent) = target.parent() {
588            std::fs::create_dir_all(parent).map_err(io_operation)?;
589        }
590        let (hash, size) = copy_backup_file(backup, relative, &target)?;
591        if size != entry.size || hash != entry.hash {
592            return Err(operation_error("backup changed during restore"));
593        }
594        set_restored_permissions(relative, &target)?;
595    }
596    Ok(())
597}

599fn create_staging_directory(destination: &Path, purpose: &str) -> Result<PathBuf, DbError> {
600    let parent = destination_parent(destination);
601    if !parent.is_dir() {
602        return Err(operation_error("destination parent is not a directory"));
603    }
604    let name = destination
605        .file_name()
606        .and_then(|value| value.to_str())
607        .ok_or_else(|| operation_error("destination name must be valid UTF-8"))?;
608    for _ in 0..16 {
609        let mut random = [0_u8; 16];
610        getrandom::fill(&mut random).map_err(|_| DbError::RandomnessUnavailable)?;
611        let candidate = parent.join(format!(
612            ".{name}.iroh-db-{purpose}-{}",
613            blake3::hash(&random).to_hex()
614        ));
615        match std::fs::create_dir(&candidate) {
616            Ok(()) => return Ok(candidate),
617            Err(error) if error.kind() == std::io::ErrorKind::AlreadyExists => {}
618            Err(error) => return Err(io_operation(error)),
619        }
620    }
621    Err(operation_error("could not allocate a staging directory"))
622}

624fn publish_staging_directory(staging: &Path, destination: &Path) -> Result<(), DbError> {
625    if destination.exists() {
626        discard_staging_directory(staging);
627        return Err(operation_error("destination appeared during publication"));
628    }
629    sync_tree_directories(staging)?;
630    if let Err(error) = std::fs::rename(staging, destination) {
631        discard_staging_directory(staging);
632        return Err(io_operation(error));
633    }
634    sync_directory(destination_parent(destination))
635}
636
637fn destination_parent(destination: &Path) -> &Path {
638    destination
639        .parent()
640        .filter(|parent| !parent.as_os_str().is_empty())
641        .unwrap_or_else(|| Path::new("."))
642}
643
644fn sync_tree_directories(root: &Path) -> Result<(), DbError> {
645    let mut directories = vec![root.to_path_buf()];
646    collect_directories(root, &mut directories)?;
647    directories.sort_by_key(|path| std::cmp::Reverse(path.components().count()));
648    for directory in directories {
649        sync_directory(&directory)?;
650    }
651    Ok(())
652}
653
654fn collect_directories(directory: &Path, directories: &mut Vec<PathBuf>) -> Result<(), DbError> {
655    for entry in std::fs::read_dir(directory).map_err(io_operation)? {
656        let entry = entry.map_err(io_operation)?;
657        if entry.file_type().map_err(io_operation)?.is_dir() {
658            let path = entry.path();
659            directories.push(path.clone());
660            collect_directories(&path, directories)?;
661        }
662    }
663    Ok(())
664}

692fn copy_and_hash(source: &Path, target: &Path) -> Result<([u8; 32], u64), DbError> {
693    let metadata = std::fs::symlink_metadata(source).map_err(io_operation)?;
694    if !metadata.file_type().is_file() || metadata.file_type().is_symlink() {
695        return Err(operation_error("backup source changed during collection"));
696    }
697    let mut reader = std::fs::File::open(source).map_err(io_operation)?;
698    let mut writer = OpenOptions::new()
699        .write(true)
700        .create_new(true)
701        .open(target)
702        .map_err(io_operation)?;
703    let result = copy_reader_and_hash(&mut reader, &mut writer)?;
704    writer.sync_all().map_err(io_operation)?;
705    Ok(result)
706}

709    backup: &Path,
710    relative: &Path,

717        .open(target)
718        .map_err(io_operation)?;

720    writer.sync_all().map_err(io_operation)?;
721    Ok(result)

724fn copy_reader_and_hash(
725    reader: &mut std::fs::File,
726    writer: &mut std::fs::File,
727) -> Result<([u8; 32], u64), DbError> {
728    let mut buffer = vec![0_u8; COPY_BUFFER_BYTES];
729    let mut hasher = blake3::Hasher::new();
730    let mut size = 0_u64;
731    loop {
732        let read = reader.read(&mut buffer).map_err(io_operation)?;
733        if read == 0 {
734            break;
735        }
736        hasher.update(&buffer[..read]);
737        writer.write_all(&buffer[..read]).map_err(io_operation)?;
738        size = size
739            .checked_add(u64::try_from(read).unwrap_or(u64::MAX))
740            .ok_or_else(|| operation_error("backup byte count overflow"))?;
741    }
742    Ok((*hasher.finalize().as_bytes(), size))
743}

759    }

761}

763fn open_backup_file(backup: &Path, relative: &Path) -> Result<std::fs::File, DbError> {
764    validate_relative(relative)?;
765    let components: Vec<_> = relative.components().collect();
766    let mut current = backup.to_path_buf();
767    for (index, component) in components.iter().enumerate() {
768        let Component::Normal(component) = component else {
769            return Err(operation_error("backup contains an unsafe path"));
770        };
771        current.push(component);
772        let metadata = std::fs::symlink_metadata(&current).map_err(io_operation)?;
773        if metadata.file_type().is_symlink()
774            || (index + 1 < components.len() && !metadata.is_dir())
775            || (index + 1 == components.len() && !metadata.is_file())
776        {
777            return Err(operation_error("backup contains a non-regular path"));
778        }
779    }
780    std::fs::File::open(current).map_err(io_operation)
781}

783fn read_bounded_regular_file(
784    root: &Path,
785    relative: &Path,
786    maximum: u64,
787) -> Result<Vec<u8>, DbError> {
788    let mut file = open_backup_file(root, relative)?;
789    let length = file.metadata().map_err(io_operation)?.len();
790    if length > maximum {
791        return Err(operation_error("backup catalog is too large"));
792    }
793    let capacity = usize::try_from(length)
794        .map_err(|_| operation_error("backup catalog exceeds platform limits"))?;
795    let mut bytes = Vec::with_capacity(capacity);
796    file.read_to_end(&mut bytes).map_err(io_operation)?;
797    Ok(bytes)
798}

808fn validate_catalog(content: &BackupCatalogContent) -> Result<(), DbError> {
809    if content.version != BACKUP_FORMAT_VERSION {
810        return Err(operation_error("unsupported backup format"));
811    }
812    if content.entries.len() > MAX_CATALOG_ENTRIES {
813        return Err(operation_error("backup catalog contains too many files"));
814    }
815    let mut previous: Option<&str> = None;
816    let mut total = 0_u64;
817    let mut has_format = false;
818    for entry in &content.entries {
819        if entry.path.len() > 4096 || entry.path == CATALOG_FILE {
820            return Err(operation_error("backup catalog contains an invalid path"));
821        }
822        let relative = Path::new(&entry.path);
823        validate_relative(relative)?;
824        if canonical_relative(relative)? != entry.path
825            || previous.is_some_and(|value| value >= entry.path.as_str())
826        {
827            return Err(operation_error(
828                "backup catalog paths are non-canonical or unordered",
829            ));
830        }
831        previous = Some(&entry.path);
832        has_format |= entry.path == "FORMAT";
833        total = total
834            .checked_add(entry.size)
835            .ok_or_else(|| operation_error("backup catalog byte count overflow"))?;
836    }
837    if !has_format {
838        return Err(operation_error("backup catalog does not contain FORMAT"));
839    }
840    Ok(())
841}
842
843fn canonical_relative(path: &Path) -> Result<String, DbError> {
844    validate_relative(path)?;
845    path.components()
846        .map(|component| match component {
847            Component::Normal(value) => value
848                .to_str()
849                .map(ToOwned::to_owned)
850                .ok_or_else(|| operation_error("backup paths must be valid UTF-8")),
851            _ => Err(operation_error("backup contains an unsafe path")),
852        })
853        .collect::<Result<Vec<_>, _>>()
854        .map(|components| components.join("/"))
855}

857fn decode_backup_canonical<T>(bytes: &[u8]) -> Result<T, DbError>
858where
859    T: minicbor::Encode<()> + for<'value> minicbor::Decode<'value, ()>,
860{
861    let mut decoder = minicbor::Decoder::new(bytes);
862    let value = decoder.decode().map_err(backup_codec)?;
863    if decoder.position() != bytes.len() || minicbor::to_vec(&value).map_err(backup_codec)? != bytes
864    {
865        return Err(operation_error("backup catalog is not canonical"));
866    }
867    Ok(value)
868}

902fn collect_files(
903    root: &Path,
904    directory: &Path,
905    options: BackupOptions,
906    files: &mut Vec<PathBuf>,
907) -> Result<(), DbError> {
908    for entry in std::fs::read_dir(directory).map_err(io_operation)? {
909        let entry = entry.map_err(io_operation)?;
910        let path = entry.path();
911        let relative = path
912            .strip_prefix(root)
913            .map_err(|_| operation_error("backup path escaped source"))?;
914        if relative.starts_with("tmp")
915            || relative == Path::new("locks/writer.lock")
916            || (!options.include_keys && relative.starts_with("keys"))
917        {
918            continue;
919        }
920        let file_type = entry.file_type().map_err(io_operation)?;
921        if file_type.is_symlink() {
922            return Err(operation_error("backup refuses symbolic links"));
923        }
924        if file_type.is_dir() {
925            collect_files(root, &path, options, files)?;
926        } else if file_type.is_file() {
927            files.push(relative.to_path_buf());
928        }
929    }
930    Ok(())
931}

933fn validate_relative(path: &Path) -> Result<(), DbError> {
934    if path.as_os_str().is_empty()
935        || path
936            .components()
937            .any(|component| !matches!(component, Component::Normal(_)))
938    {
939        return Err(operation_error("backup contains an unsafe path"));
940    }
941    Ok(())
942}
