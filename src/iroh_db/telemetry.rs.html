6/// Aggregate latency evidence without high-cardinality labels.
7#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
8pub struct LatencyMetrics {
9    samples: u64,
10    failures: u64,
11    total: Duration,
12    maximum: Duration,
13}
14
15impl LatencyMetrics {
16    /// Number of completed operations observed.
17    pub const fn samples(self) -> u64 {
18        self.samples
19    }
20
21    /// Number of observed operations that returned an error.
22    pub const fn failures(self) -> u64 {
23        self.failures
24    }
25
26    /// Sum of all observed operation durations.
27    pub const fn total(self) -> Duration {
28        self.total
29    }
30
31    /// Largest observed operation duration.
32    pub const fn maximum(self) -> Duration {
33        self.maximum
34    }
35}
36
37#[derive(Debug, Default)]
38struct LatencyRecorder {
39    samples: AtomicU64,
40    failures: AtomicU64,
41    total_nanos: AtomicU64,
42    maximum_nanos: AtomicU64,
43}
44
45impl LatencyRecorder {
46    fn observe(&self, elapsed: Duration, failed: bool) {
47        let nanos = u64::try_from(elapsed.as_nanos()).unwrap_or(u64::MAX);
48        saturating_add(&self.samples, 1);
49        if failed {
50            saturating_add(&self.failures, 1);
51        }
52        saturating_add(&self.total_nanos, nanos);
53        self.maximum_nanos.fetch_max(nanos, Ordering::Relaxed);
54    }
55
56    fn snapshot(&self) -> LatencyMetrics {
57        LatencyMetrics {
58            samples: self.samples.load(Ordering::Relaxed),
59            failures: self.failures.load(Ordering::Relaxed),
60            total: Duration::from_nanos(self.total_nanos.load(Ordering::Relaxed)),
61            maximum: Duration::from_nanos(self.maximum_nanos.load(Ordering::Relaxed)),
62        }
63    }
64}

66#[derive(Debug, Default)]
67pub(crate) struct Telemetry {
68    local_apply: LatencyRecorder,
69    query: LatencyRecorder,
70    index_records_scanned: AtomicU64,
71    equivocation_quarantines: AtomicU64,
72    schema_rebases: AtomicU64,
73    sync_rounds: AtomicU64,
74    sync_failures: AtomicU64,
75    sync_received: AtomicU64,
76    sync_sent: AtomicU64,
77    snapshots_created: AtomicU64,
78    snapshots_installed: AtomicU64,
79    repairs_succeeded: AtomicU64,
80    repairs_failed: AtomicU64,
81    last_repair_consistency: AtomicU8,
82}
83
84impl Telemetry {
85    pub(crate) fn observe_local_apply(&self, elapsed: Duration, failed: bool) {
86        self.local_apply.observe(elapsed, failed);
87    }
88
89    pub(crate) fn observe_query(&self, elapsed: Duration, failed: bool) {
90        self.query.observe(elapsed, failed);
91    }
92
93    pub(crate) fn add_index_records_scanned(&self, records: usize) {
94        saturating_add(
95            &self.index_records_scanned,
96            u64::try_from(records).unwrap_or(u64::MAX),
97        );
98    }
99
100    pub(crate) fn record_equivocation_quarantine(&self) {
101        saturating_add(&self.equivocation_quarantines, 1);
102    }
103
104    pub(crate) fn record_schema_rebase(&self) {
105        saturating_add(&self.schema_rebases, 1);
106    }
107
108    pub(crate) fn record_sync(&self, received: usize, sent: usize, failed: bool) {
109        saturating_add(&self.sync_rounds, 1);
110        if failed {
111            saturating_add(&self.sync_failures, 1);
112        }
113        saturating_add(
114            &self.sync_received,
115            u64::try_from(received).unwrap_or(u64::MAX),
116        );
117        saturating_add(&self.sync_sent, u64::try_from(sent).unwrap_or(u64::MAX));
118    }
119
120    pub(crate) fn record_snapshot_created(&self) {
121        saturating_add(&self.snapshots_created, 1);
122    }
123
124    pub(crate) fn record_snapshot_installed(&self) {
125        saturating_add(&self.snapshots_installed, 1);
126    }
127
128    pub(crate) fn record_repair(&self, consistent: Option<bool>) {
129        if let Some(consistent) = consistent {
130            saturating_add(&self.repairs_succeeded, 1);
131            self.last_repair_consistency
132                .store(if consistent { 2 } else { 1 }, Ordering::Relaxed);
133        } else {
134            saturating_add(&self.repairs_failed, 1);
135            self.last_repair_consistency.store(0, Ordering::Relaxed);
136        }
137    }
138
139    pub(crate) fn snapshot(&self) -> RuntimeMetrics {
140        RuntimeMetrics {
141            local_apply: self.local_apply.snapshot(),
142            query: self.query.snapshot(),
143            index_records_scanned: self.index_records_scanned.load(Ordering::Relaxed),
144            equivocation_quarantines: self.equivocation_quarantines.load(Ordering::Relaxed),
145            schema_rebases: self.schema_rebases.load(Ordering::Relaxed),
146            sync_rounds: self.sync_rounds.load(Ordering::Relaxed),
147            sync_failures: self.sync_failures.load(Ordering::Relaxed),
148            sync_received: self.sync_received.load(Ordering::Relaxed),
149            sync_sent: self.sync_sent.load(Ordering::Relaxed),
150            snapshots_created: self.snapshots_created.load(Ordering::Relaxed),
151            snapshots_installed: self.snapshots_installed.load(Ordering::Relaxed),
152            repairs_succeeded: self.repairs_succeeded.load(Ordering::Relaxed),
153            repairs_failed: self.repairs_failed.load(Ordering::Relaxed),
154            last_repair_derived_consistent: match self
155                .last_repair_consistency
156                .load(Ordering::Relaxed)
157            {
158                1 => Some(false),
159                2 => Some(true),
160                _ => None,
161            },
162        }
163    }
164}

166fn saturating_add(counter: &AtomicU64, value: u64) {
167    let _previous = counter.fetch_update(Ordering::Relaxed, Ordering::Relaxed, |current| {
168        Some(current.saturating_add(value))
169    });
170}
